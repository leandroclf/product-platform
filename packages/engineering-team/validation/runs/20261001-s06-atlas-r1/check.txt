import rates


def quote_order(lines):
    """lines: iterable of (amount, currency). Returns total in USD."""
    quote_rates = {}

    def converted_amounts():
        for amount, cur in lines:
            if cur not in quote_rates:
                quote_rates[cur] = rates.get_rate(cur)
            yield amount * quote_rates[cur]

    return round(sum(converted_amounts()), 2)
