import rates


def quote_order(lines):
    """lines: iterable of (amount, currency). Returns total in USD."""
    rates_by_currency = {}

    def converted_amounts():
        for amount, currency in lines:
            if currency not in rates_by_currency:
                rates_by_currency[currency] = rates.get_rate(currency)
            yield amount * rates_by_currency[currency]

    return round(sum(converted_amounts()), 2)
