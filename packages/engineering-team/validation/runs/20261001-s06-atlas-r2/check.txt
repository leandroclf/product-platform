import rates


def quote_order(lines):
    """lines: iterable of (amount, currency). Returns total in USD."""
    quote_rates = {}

    def get_quote_rate(currency):
        # Reuse rates only within this quote so later quotes see fresh values.
        if currency not in quote_rates:
            quote_rates[currency] = rates.get_rate(currency)
        return quote_rates[currency]

    return round(sum(amount * get_quote_rate(cur) for amount, cur in lines), 2)
